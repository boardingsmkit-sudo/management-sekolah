-- Disiplin KBM: kategori pelanggaran per catatan presensi + pengaturan bobot. Jalankan sekali di Supabase SQL Editor. -- Asumsi (dari kode aplikasi, mohon diperiksa): profiles.id = auth.uid() dan profiles.role berisi 'Super Admin', 'Kepala Sekolah', dst. -- Tidak ada data lama yang diubah atau dihapus. -- 1) Kolom kategori pada presensi KBM (daftar kunci, mis. ["terlambat","tidak_rapi"]). Nullable; baris lama tetap kosong. alter table public.attendance_kbm add column if not exists kategori_pelanggaran jsonb default '[]'::jsonb; -- 2) Pengaturan bobot (nilai awal, batas minimum, bobot per kategori), pola sama dengan boarding_settings. create table if not exists public.kbm_settings ( key text primary key, value jsonb not null, updated_at timestamptz not null default now() ); alter table public.kbm_settings enable row level security; drop policy if exists kbm_settings_select on public.kbm_settings; create policy kbm_settings_select on public.kbm_settings for select to authenticated using (true); drop policy if exists kbm_settings_write on public.kbm_settings; create policy kbm_settings_write on public.kbm_settings for all to authenticated using (exists (select 1 from public.profiles p where p.id = auth.uid() and p.role in ('Super Admin','Kepala Sekolah'))) with check (exists (select 1 from public.profiles p where p.id = auth.uid() and p.role in ('Super Admin','Kepala Sekolah')));
